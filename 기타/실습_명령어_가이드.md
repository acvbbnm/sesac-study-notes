print(int("123") + 1, str(456) + "원")

\# 연산자

print(7 // 3, 7 % 3, 2 ** 10) \# 몫, 나머지, 거듭제곱

\# 조건문

score = 85

if score >= 90: print("A")

elif score >= 80: print("B")

else: print("C")

\# 반복문 + range

for i in range(1, 6):

print(i, end=" ") \# 1 2 3 4 5

\# while + break/continue

n = 0

while True:

n += 1

if n == 3: continue \# 3 건너뛰기

if n > 5: break \# 5 넘으면 종료

print(n, end=" ") \# 1 2 4 5

\# 리스트

nums = [10, 20, 30, 40]

print(nums[0], nums[-1], nums[1:3]) \# 10 40 [20,30]

nums.append(50); nums.remove(20)

print(sum(nums), max(nums), len(nums))

print([x**2 for x in range(1,5)]) \# 컴프리헨션 [1,4,9,16]

\# 딕셔너리

person = {"name":"Alice", "age":30}

print(person["name"], person.get("email","없음"))

person["city"] = "Seoul"

for k, v in person.items(): print(k, v)

\# 함수

def add(a, b=10): \# b는 기본값

return a + b

print(add(3), add(3, 5)) \# 13 8

def stats(*args): \# 가변인자 → 튜플

return sum(args), max(args)

print(stats(4, 1, 9, 2)) \# (16, 9)

def profile(**kw): \# 키워드 가변인자 → 딕셔너리

for k, v in kw.items(): print(k, v)

profile(name="Bob", job="dev")

3. 판다스 — 선택 · 필터

titanic["Age"] \# 한 열 (Series)

titanic[["Age","Sex"]] \# 여러 열 (DataFrame, 대괄호 2개)

\# 조건 필터

titanic[titanic["Age"] > 35] \# 35세 초과

titanic[titanic["Pclass"].isin([2,3])] \# 2 또는 3등급

titanic[(titanic["Age"]>35) & (titanic["Sex"]=="female")] \# &,\| + 소괄호

titanic[titanic["Age"].notna()] \# 결측 아닌 행

\# loc(이름) vs iloc(위치)

titanic.loc[titanic["Age"]>35, "Name"] \# 조건 + 열 이름

titanic.iloc[0:5, 2:5] \# 0~4행, 2~4열

X_tr, X_val, y_tr, y_val = train_test_split(

8. 자주 쓰는 명령어 빠른 참조

| **명령어**                         | **기능**                |
|------------------------------------|-------------------------|
| pd.read_csv(url)                   | CSV 불러오기            |
| df.head() / .tail()                | 앞/뒤 미리보기          |
| df.info() / .describe()            | 구조·결측치 / 통계 요약 |
| df.shape                           | 행·열 개수 (속성)       |
| df[df[c] > 0]                 | 조건으로 행 필터        |
| df.loc[행,열] / .iloc[i,j]     | 이름 / 위치 기반 선택   |
| df.sort_values(by=...)             | 정렬                    |
| df.groupby(c)[c2].mean()         | 그룹별 집계             |
| df[c].value_counts()             | 값별 개수               |
| pd.get_dummies(df, columns=...)    | 원-핫 인코딩            |
| pd.cut(x, bins=..., labels=...)    | 연속값 구간화           |
| train_test_split(X, y, test_size=) | 학습/평가 분할          |
| StandardScaler().fit_transform(X)  | 표준화                  |
| SimpleImputer(strategy='mean')     | 결측치 평균 대체        |
| np.log1p(x)                        | 로그 변환 log(x+1)      |
| sns.boxplot(data=, x=, y=)         | 박스플롯(분포·이상치)   |
